วิธีเชื่อม Google Sheets ของคุณเอง (ใช้เวลา 1 นาที):

  1. สร้างไฟล์ Google Sheets ใหม่ ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. คัดลอกโค้ดด้านล่างไปวางทับ แล้วกด การทำให้ใช้งานได้ (Deploy) > รายการใหม่ > เว็บแอป (Web App) เลือกผู้มีสิทธิ์เข้าถึงเป็น "ทุกคน (Anyone)"
  3. คัดลอก URL ที่ได้มาวางในช่องด้านบน ข้อมูลจะถูกส่งเข้า Google Sheets อัตโนมัติทุกครั้งที่กดบันทึก!
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "JobID", "Status", "ServiceDate", "MachineCode", "MachineName", "WorkOrder", "Technician", "Supervisor", "Temp1_C", "Temp2_C", "Temp3_C", "Vib1_mms", "Vib2_mms", "Vib3_mms", "AlarmCode", "Comment", "TotalPages"]);
  }
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), d.jobId, d.status, d.serviceDate, d.machineCode, d.machineName, d.workOrder, d.technician, d.supervisor, d.temp1, d.temp2, d.temp3, d.vib1, d.vib2, d.vib3, d.alarmCode, d.comment, d.totalPages]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
